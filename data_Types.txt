cart = [
    {"product": "Shirt", "price": 599, "qty": 2},
    {"product": "Shoes", "price": 1299, "qty": 1},
    {"product": "Cap", "price": 199, "qty": 3}
]


empty=[]
discount=10
if sum(empty) >1000:
    sum(empty)*discount/100
for item in cart:
    product=item["product"]
    price=item["price"]
    qty=item["qty"]
    sales=price*qty
    empty.append(sales)
    #grand_total=sum(sales)
    print(f"{product}: {price} x {qty} = {sales}")
    #print(grand_total)
print("Grand Total:",sum(empty))
print("Discount Applied (10%):",sum(empty)*discount/100)
print("Final Payable Amount:",sum(empty)-sum(empty)*discount/100)
